WITH filtered_edges AS (
    SELECT
        t."UZ_Osnovnoj",
        t."UZ_Priemnik",
        t."Znachenie",
        t."Tip_svjazi",
        t."Gorod",
        -- Свойства узла-приёмника (как было раньше)
        (
            SELECT p."Znachenie"
            FROM "xls_medintsev_dataset_graf" p
            WHERE p."UZ_Osnovnoj" = t."UZ_Priemnik"
            LIMIT 1
        ) AS "Znachenie_UZ_Priemnik",
        (
            SELECT p."Gorod"
            FROM "xls_medintsev_dataset_graf" p
            WHERE p."UZ_Osnovnoj" = t."UZ_Priemnik"
            LIMIT 1
        ) AS "Gorod_UZ_Priemnik"
    FROM "xls_medintsev_dataset_graf" t
    WHERE 
        -- Здесь используется переменная фильтра
        (t."UZ_Osnovnoj" IN ({!env.uz_filter!; ''}) OR {!env.uz_filter!; ''} = '')
        -- Если фильтр по другим полям (например Gorod), добавьте сюда свои условия
),

-- Все узлы, которые должны быть в графе после фильтра
all_nodes AS (
    SELECT DISTINCT "UZ_Osnovnoj" AS node FROM filtered_edges
    UNION
    SELECT DISTINCT "UZ_Priemnik" AS node FROM filtered_edges 
    WHERE "UZ_Priemnik" IS NOT NULL
),

-- Недостающие узлы (те, что есть только как «Потомок»)
missing_nodes AS (
    SELECT 
        n.node AS "UZ_Osnovnoj",
        NULL::text AS "UZ_Priemnik",           -- dummy, чтобы не создавать лишних связей
        NULL::int  AS "Znachenie",
        NULL::text AS "Tip_svjazi",
        (
            SELECT p."Gorod"
            FROM "xls_medintsev_dataset_graf" p
            WHERE p."UZ_Osnovnoj" = n.node
            LIMIT 1
        ) AS "Gorod",
        -- Свойства самого узла (на случай, если визуализация их использует)
        (
            SELECT p."Znachenie"
            FROM "xls_medintsev_dataset_graf" p
            WHERE p."UZ_Osnovnoj" = n.node
            LIMIT 1
        ) AS "Znachenie_UZ_Priemnik",
        (
            SELECT p."Gorod"
            FROM "xls_medintsev_dataset_graf" p
            WHERE p."UZ_Osnovnoj" = n.node
            LIMIT 1
        ) AS "Gorod_UZ_Priemnik"
    FROM all_nodes n
    LEFT JOIN filtered_edges f ON f."UZ_Osnovnoj" = n.node
    WHERE f."UZ_Osnovnoj" IS NULL
)

-- Финальный результат: отфильтрованные связи + недостающие узлы
SELECT * FROM filtered_edges
UNION ALL
SELECT * FROM missing_nodes
ORDER BY "UZ_Osnovnoj", "UZ_Priemnik" NULLS LAST;
